# LOI_V_WH_OVERVIEW — 仓库概览物化视图

> 收集时间: 2026-05-27
> 刷新策略: REFRESH COMPLETE ON DEMAND, 每4小时 (SYSDATE + 4/24)
> 状态: FRESH ✓

## 业务逻辑

仓库概览视图，统计各仓库/库位类型的数量和存储天数分布。

### 数据源
- LOI_V_WAREHOUSE_CAPACITY_BASE — 仓库容量基础数据
- LOI_V_SUB_INVENTORY_ALL_LATEST — 最新库存快照
- LOI_MT_WAREHOUSE_DEFINITION — 仓库定义表

### 库位计数规则

| 库位类型 | SU 类型 | 计数规则 |
|---------|---------|---------|
| AR1 | K3/K4/K5 | COUNT(HU) × 2 |
| AR1 | 其他 | COUNT(HU) |
| V02 | - | COUNT(STORAGE_BIN) |
| 其他 | - | COUNT(STORAGE_BIN) |

### 存储天数分段
- 0~30 DAYS
- 31~90 DAYS
- 91~180 DAYS
- 181~365 DAYS
- 366~730 DAYS
- Over 730 DAYS (310 类型特殊处理为 366~730)

### 特殊逻辑
1. RDC-WX 仓库单独处理（从 LOI_V_WAREHOUSE_CAPACITY_BASE 取，过滤 warehouse = 'RDC-WX'）
2. E682 工厂的 8H/8B/8F 开头的库位类型也计入 RDC-WX
3. 去重规则：按 GOODS_RECEIPT_DATE + STORAGE_TYPE + STORAGE_BIN + HU 分区，取第一条

## SQL 定义

```sql
CREATE MATERIALIZED VIEW "LOGPOE"."LOI_V_WH_OVERVIEW" 
  ("WAREHOUSE", "STORAGE_TYPE", "AMOUNT", "STORAGE_DAYS", "MAXCAP")
REFRESH COMPLETE ON DEMAND START WITH sysdate+0 NEXT SYSDATE + 4/24
AS
WITH BASE AS (
  -- 1. 正常仓库：除 RDC-WX 外
  SELECT A.WAREHOUSE, A.STORAGE_TYPE, A.HU, A.SU, B.MAX_OCCUPIED_CAPACITY, A.STORAGE_BIN,
    CASE
      WHEN ROUND(TRUNC(SYSDATE) - TO_DATE(GOODS_RECEIPT_DATE,'YYYYMMDD')) <= 30 THEN '0~30 DAYS'
      WHEN ROUND(TRUNC(SYSDATE) - TO_DATE(GOODS_RECEIPT_DATE,'YYYYMMDD')) BETWEEN 31 AND 90 THEN '31~90 DAYS'
      WHEN ROUND(TRUNC(SYSDATE) - TO_DATE(GOODS_RECEIPT_DATE,'YYYYMMDD')) BETWEEN 91 AND 180 THEN '91~180 DAYS'
      WHEN ROUND(TRUNC(SYSDATE) - TO_DATE(GOODS_RECEIPT_DATE,'YYYYMMDD')) BETWEEN 181 AND 365 THEN '181~365 DAYS'
      WHEN ROUND(TRUNC(SYSDATE) - TO_DATE(GOODS_RECEIPT_DATE,'YYYYMMDD')) BETWEEN 366 AND 730 THEN '366~730 DAYS'
      WHEN ROUND(TRUNC(SYSDATE) - TO_DATE(GOODS_RECEIPT_DATE,'YYYYMMDD')) > 730 AND A.STORAGE_TYPE = '310' THEN '366~730 DAYS'
      WHEN ROUND(TRUNC(SYSDATE) - TO_DATE(GOODS_RECEIPT_DATE,'YYYYMMDD')) > 730 THEN 'Over 730 DAYS'
    END STORAGE_DAYS
  FROM (
    SELECT t.* FROM (
      SELECT a.*, RANK() OVER (
        PARTITION BY a.GOODS_RECEIPT_DATE||a.STORAGE_TYPE||a.STORAGE_BIN||a.HU
        ORDER BY a.GOODS_RECEIPT_DATE, a.STORAGE_BIN, a.HU ASC
      ) rn
      FROM LOI_V_WAREHOUSE_CAPACITY_BASE a
      WHERE a.GOODS_RECEIPT_DATE <> '00000000' AND a.WAREHOUSE <> 'RDC-WX'
    ) t WHERE rn = 1
  ) A
  LEFT JOIN LOI_MT_WAREHOUSE_DEFINITION B
  ON A.STORAGE_TYPE = B.STORAGE_TYPE AND A.WAREHOUSE = B.WAREHOUSE

  UNION ALL

  -- 2. RDC-WX 仓库
  SELECT A.WAREHOUSE, A.STORAGE_TYPE, A.HU, A.SU, B.MAX_OCCUPIED_CAPACITY, A.STORAGE_BIN,
    CASE
      WHEN ROUND(TRUNC(SYSDATE) - TO_DATE(GOODS_RECEIPT_DATE,'YYYYMMDD')) <= 30 THEN '0~30 DAYS'
      -- ... 同上分段逻辑
    END STORAGE_DAYS
  FROM (
    SELECT t.* FROM (
      SELECT a.*, RANK() OVER (
        PARTITION BY a.GOODS_RECEIPT_DATE||a.STORAGE_TYPE||a.STORAGE_BIN||a.HU
        ORDER BY a.GOODS_RECEIPT_DATE, a.STORAGE_BIN, a.HU ASC
      ) rn
      FROM LOI_V_WAREHOUSE_CAPACITY_BASE a
      WHERE a.GOODS_RECEIPT_DATE <> '00000000' AND a.WAREHOUSE = 'RDC-WX'
    ) t WHERE rn = 1
  ) A
  LEFT JOIN LOI_MT_WAREHOUSE_DEFINITION B
  ON A.STORAGE_TYPE = B.STORAGE_TYPE AND A.WAREHOUSE = B.WAREHOUSE

  UNION ALL

  -- 3. E682 的 8H/8B/8F 库位计入 RDC-WX
  SELECT 'RDC-WX' AS WAREHOUSE, s.STORAGE_TYPE, NULL AS HU, NULL AS SU, 
    b.MAX_OCCUPIED_CAPACITY, s.STORAGE_BIN, 
    CASE ... END AS STORAGE_DAYS
  FROM (
    SELECT t.* FROM (
      SELECT s.*, RANK() OVER (
        PARTITION BY s.GOODS_RECEIPT_DATE||s.STORAGE_TYPE||s.STORAGE_BIN
        ORDER BY s.GOODS_RECEIPT_DATE, s.STORAGE_BIN
      ) rn
      FROM LOI_V_SUB_INVENTORY_ALL_LATEST s
      WHERE s.PLANT = 'E682' AND s.RECORD_TYPE = 'STOCK_ON_HAND' AND s.IS_LATEST = 'Y'
        AND (s.STORAGE_TYPE LIKE '8H%' OR s.STORAGE_TYPE LIKE '8F%' OR s.STORAGE_TYPE LIKE '8B%')
        AND s.GOODS_RECEIPT_DATE <> '00000000'
    ) t WHERE t.rn = 1
  ) s
  LEFT JOIN LOI_MT_WAREHOUSE_DEFINITION b
  ON b.WAREHOUSE = 'RDC-WX' AND b.STORAGE_TYPE = s.STORAGE_TYPE
)
-- 汇总
SELECT WAREHOUSE, STORAGE_TYPE, SUM(AMOUNT) AS AMOUNT, STORAGE_DAYS, MAXCAP
FROM (
  -- 普通仓位（除 V02、AR1 外）
  SELECT WAREHOUSE, STORAGE_TYPE, COUNT(STORAGE_BIN) AS AMOUNT, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY AS MAXCAP
  FROM BASE WHERE STORAGE_TYPE NOT IN ('V02','AR1')
  GROUP BY WAREHOUSE, STORAGE_TYPE, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY

  UNION ALL

  -- V02：RDC-WX2 单独统计
  SELECT WAREHOUSE, STORAGE_TYPE, COUNT(STORAGE_BIN) AS AMOUNT, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY AS MAXCAP
  FROM BASE WHERE STORAGE_TYPE = 'V02' AND WAREHOUSE = 'RDC-WX2'
  GROUP BY WAREHOUSE, STORAGE_TYPE, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY

  UNION ALL

  -- V02：其他仓库
  SELECT WAREHOUSE, STORAGE_TYPE, COUNT(STORAGE_BIN) AS AMOUNT, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY AS MAXCAP
  FROM BASE WHERE STORAGE_TYPE = 'V02' AND WAREHOUSE != 'RDC-WX2'
  GROUP BY WAREHOUSE, STORAGE_TYPE, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY

  UNION ALL

  -- AR1：普通托盘 SU ≠ K3/K4/K5，按 HU 计数
  SELECT WAREHOUSE, STORAGE_TYPE, COUNT(HU) AS AMOUNT, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY AS MAXCAP
  FROM BASE WHERE STORAGE_TYPE = 'AR1' AND SU NOT IN ('K3','K4','K5')
  GROUP BY WAREHOUSE, STORAGE_TYPE, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY

  UNION ALL

  -- AR1：特殊托盘 SU ∈ K3/K4/K5，按 HU×2 计数
  SELECT WAREHOUSE, STORAGE_TYPE, COUNT(HU) * 2 AS AMOUNT, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY AS MAXCAP
  FROM BASE WHERE STORAGE_TYPE = 'AR1' AND SU IN ('K3','K4','K5')
  GROUP BY WAREHOUSE, STORAGE_TYPE, STORAGE_DAYS, MAX_OCCUPIED_CAPACITY
)
GROUP BY WAREHOUSE, STORAGE_TYPE, STORAGE_DAYS, MAXCAP;
```
